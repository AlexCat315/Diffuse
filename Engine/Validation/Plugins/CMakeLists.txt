cmake_minimum_required(VERSION 3.20)
project(cove-plugin-probe LANGUAGES C)
if(NOT DEFINED COVE_WASMTIME_VARIANT)
    message(FATAL_ERROR "Set COVE_WASMTIME_VARIANT to a staged artifact variant directory")
endif()
add_executable(CovePluginProbe Probe.c ../../Sources/Bridge/CWasmtimeBridge/CWasmtimeBridge.c)
target_include_directories(CovePluginProbe PRIVATE
    ../../Sources/Bridge/CWasmtimeBridge/include "${COVE_WASMTIME_VARIANT}/include")
target_compile_definitions(CovePluginProbe PRIVATE WASM_API_EXTERN= WASI_API_EXTERN=)
if(WIN32)
    target_link_libraries(CovePluginProbe PRIVATE "${COVE_WASMTIME_VARIANT}/lib/wasmtime.lib"
        ws2_32 advapi32 userenv ntdll shell32 ole32 bcrypt)
else()
    target_link_libraries(CovePluginProbe PRIVATE "${COVE_WASMTIME_VARIANT}/lib/libwasmtime.a")
    if(APPLE)
        target_link_libraries(CovePluginProbe PRIVATE "-framework CoreFoundation")
    elseif(ANDROID)
        target_link_libraries(CovePluginProbe PRIVATE dl m)
        target_link_options(CovePluginProbe PRIVATE "-Wl,-z,max-page-size=16384")
    else()
        target_link_libraries(CovePluginProbe PRIVATE pthread dl m)
    endif()
endif()
