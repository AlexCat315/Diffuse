cmake_minimum_required(VERSION 3.20)
project(guava-plugin-probe LANGUAGES C)
if(NOT DEFINED GUAVA_WASMTIME_VARIANT)
    message(FATAL_ERROR "Set GUAVA_WASMTIME_VARIANT to a staged artifact variant directory")
endif()
add_executable(GuavaPluginProbe Probe.c ../../Sources/Bridge/CWasmtimeBridge/CWasmtimeBridge.c)
target_include_directories(GuavaPluginProbe PRIVATE
    ../../Sources/Bridge/CWasmtimeBridge/include "${GUAVA_WASMTIME_VARIANT}/include")
target_compile_definitions(GuavaPluginProbe PRIVATE WASM_API_EXTERN= WASI_API_EXTERN=)
if(WIN32)
    target_link_libraries(GuavaPluginProbe PRIVATE "${GUAVA_WASMTIME_VARIANT}/lib/wasmtime.lib"
        ws2_32 advapi32 userenv ntdll shell32 ole32 bcrypt)
else()
    target_link_libraries(GuavaPluginProbe PRIVATE "${GUAVA_WASMTIME_VARIANT}/lib/libwasmtime.a")
    if(APPLE)
        target_link_libraries(GuavaPluginProbe PRIVATE "-framework CoreFoundation")
    elseif(ANDROID)
        target_link_libraries(GuavaPluginProbe PRIVATE dl m)
        target_link_options(GuavaPluginProbe PRIVATE "-Wl,-z,max-page-size=16384")
    else()
        target_link_libraries(GuavaPluginProbe PRIVATE pthread dl m)
    endif()
endif()
